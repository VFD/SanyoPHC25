; Ce qui se trouve à la fin de la ROM 1.3
; 
; Du code source addembleur.
; Ce code source ne sert à rien.
; 
; Il sera remplacer par du code Z80 pour la v1.4.
;

CFCERR
	OR	D
	JR	Z,CFCERR
	LD	A,D
	OR	A
	JR	NZ,CFCERR
	LD	L,3
	CALL	FSDTBL
	LD	(HL),E
	RET

TSOUND:	JR	C,TSND0
	LD	E,120
TSND0:	LD	A,E
	CP	32
	JR	C,CFCERR
	LD	A,D
	OR	A
	JR	NZ,CFCERR
	LD	L,4		;TEMPO
	CALL	FSDTBL
	LD	(HL),E
	RET

RSOUND:	JR	C,RSND0
	LD	E,4
RSND0:	LD	A,E
	CP	65
	JR	NC,CFCERR
	OR	D
	JR	Z,CFCERR
	LD	A,D
	OR	A
	JR	NZ,CFCERR
RSND1:	LD	HL,0
	PUSH	HL
	LD	L,5
	CALL	FSDTBL		;VOL.
	LD	A,(HL)
	LD	(LSNDST),A
	LD	(HL),0
	JP	OCTX0

NSOUND:	JR	NC,CFCERR
NSND0:	LD	A,D
	OR	A
	JR	NZ,CFCERR
	LD	A,E
	OR	A
	JR	Z,NSND1
	CP	97
	JR	NC,CFCERR
	LD	C,E
	LD	A,12
	CALL	BITDEV
	INC	C
	LD	E,C		;OCTAVE
	ADD	A,A
	LD	B,0
	LD	C,A		;OFFSET
	JP	OCTSET
	;
NSND1:	CALL	FTSTDT
	LD	DE,4
	JR	Z,RSND0
	CALL	STRPT1
	JR	RSND0

INTSND:	PUSH	BC
	PUSH	AF
	PUSH	DE
	PUSH	HL
	LD	A,(REGNO)
	LD	C,A
	XOR	A
INTSN3:	RR	C
	PUSH	BC
	PUSH	AF
	CALL	C,SNDPUT
	POP	AF
	POP	BC
	INC	A
	CP	3
	JR	C,INTSN3
	JP	ALLPOP

SNDPUT:	LD	B,A		;REG.NO.
	LD	L,1		;0,1,2
	CALL	FSDTB0
	LD	D,(HL)
	DEC	HL
